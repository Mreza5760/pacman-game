cmake_minimum_required(VERSION 3.15)
project(Pacman C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/build)
set(BUILD_SHARED_LIBS OFF)

set(SOURCES
    src/main.c
    src/map.c
    src/menu.c
    src/draw.c
    src/ghost.c
    src/player.c
    src/record.c
    src/embedded_loader.c
    src/embedded_assets.c
)

add_executable(${PROJECT_NAME} ${SOURCES})

# Manual static linking for Windows
set(RAYLIB_PATH $ENV{RAYLIB_PATH})
target_include_directories(${PROJECT_NAME} PRIVATE ${RAYLIB_PATH}/include)
target_link_directories(${PROJECT_NAME} PRIVATE ${RAYLIB_PATH}/lib)
target_link_libraries(${PROJECT_NAME} PRIVATE raylib winmm gdi32)

set_target_properties(${PROJECT_NAME} PROPERTIES WIN32_EXECUTABLE TRUE)
message(STATUS "Project: ${PROJECT_NAME}")
message(STATUS "Output Directory: ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")

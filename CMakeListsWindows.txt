cmake_minimum_required(VERSION 3.15)
project(Pacman C)
set(CMAKE_C_STANDARD 23)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/build)
set(BUILD_SHARED_LIBS OFF)

set(SOURCES
    src/main.c
    src/map.c
    src/menu.c
    src/draw.c
    src/ghost.c
    src/player.c
    src/record.c
    src/embedded_loader.c
    src/embedded_assets.c
)

add_executable(${PROJECT_NAME} ${SOURCES})

# Point to your locally built raylib
set(RAYLIB_PATH "C:/Users/User/raylib")
target_include_directories(${PROJECT_NAME} PRIVATE ${RAYLIB_PATH}/src)
target_link_directories(${PROJECT_NAME} PRIVATE ${RAYLIB_PATH}/build/raylib)
target_link_libraries(${PROJECT_NAME} PRIVATE raylib winmm gdi32 opengl32 user32 shell32)

# For MinGW/GCC: use -e mainCRTStartup instead of /ENTRY
set_target_properties(${PROJECT_NAME} PROPERTIES
    WIN32_EXECUTABLE TRUE
    LINK_FLAGS "-e mainCRTStartup"
)

message(STATUS "Project: ${PROJECT_NAME}")
message(STATUS "Output Directory: ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
